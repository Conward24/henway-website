Build WaitlistMailer. (Bolt builds full web apps right in the browser.)
Upload a spreadsheet of names and emails, then the tool fills in a personal template for each person and sends emails in small batches to protect inbox delivery. Key features: Spreadsheet Upload, Smart Batching, Personal Merge Fields, Unsubscribe Link, Delivery Tracking.

## Your job: build these for real (this is why we came to you)
The reference app below deliberately stops short of these, because they need their own backend. Build each one properly. Do not stand one in with a share sheet, a copy button or mock data.
- **Smart Batching / automatic small-batch sending to protect inbox delivery** — actually dispatching emails in throttled batches requires a real mail-sending service (SMTP/API) with rate control; a static file cannot send email on its own or in the background _(needs: integration, automatic sending)_
- **Automated Delivery Tracking (opens, bounces, deliverability)** — real delivery/open/bounce tracking requires a mail server or ESP webhook pipeline reporting back live status, which no local file can observe _(needs: integration, live data)_
- **Automatic Unsubscribe processing** — honoring unsubscribes across future sends needs a persistent, server-side suppression list shared across sends, not just a link in a single device's browser _(needs: database, integration)_

## Setup the person will need (they do not write code)
- An account with the outside service this needs (for texts, an SMS provider; for payments, a payment processor): the person signs up, usually pays per message or per charge, and gets keys for you to connect.
- A real shared database, so everyone sees the same data and the first answer wins on the server, not in one person's browser.
- Walk them through each step when you get to it: what to sign up for, what it costs, and exactly which value goes where. Secret keys go in your settings, never in the chat.

## Already working (see the reference app below)
A working version of this part already exists. Keep its layout and what it does well. Where the reference says it cannot do something from "Your job" above, that is the gap you are closing, not a rule to keep.
- Spreadsheet Upload — parse a CSV file client-side (e.g. via JS CSV parsing) into a table of names/emails stored locally
- Personal Merge Fields — template editor that substitutes {{name}} style placeholders per row, previewed in-browser
- Delivery Tracking (manual log) — a local log/tracker where the user marks each recipient as sent/opened/bounced after the fact, stored in browser storage

- Unsubscribe Link — delivered by mailto
- Sending the emails — delivered by mailto
- Save/reload recipient list — delivered by data_backup

/* henwayStore — a working localStorage inside a sandboxed, opaque-origin
   document, where the real one throws SecurityError on contact.
   Installed before any app code runs. Never throws. */
window.henwayStore = (function () {
  var seed = window.__henwayStore || {};

  var native = null;
  try {
    window.localStorage.setItem('__henway_probe', '1');
    window.localStorage.removeItem('__henway_probe');
    native = window.localStorage;
  } catch (e) { native = null; }

  if (native) {
    /* Downloaded file, or any context where storage really works. Fold in the
       seed once so a hosted->downloaded move keeps the data, then get out of
       the way. */
    try {
      for (var sk in seed) {
        if (Object.prototype.hasOwnProperty.call(seed, sk) && native.getItem(sk) === null) {
          native.setItem(sk, String(seed[sk]));
        }
      }
    } catch (e2) {}
    return { mode: 'native', flush: function () {}, backed: true };
  }

  var mem = {};
  for (var k in seed) {
    if (Object.prototype.hasOwnProperty.call(seed, k)) mem[k] = String(seed[k]);
  }

  var timer = null, dirty = false;
  function schedule() {
    dirty = true;
    clearTimeout(timer);
    timer = setTimeout(flush, 700);            /* coalesce rapid edits */
  }
  /* The app hands its data OUT and never persists it itself. Whoever is hosting
     the frame decides where it goes: the shell keeps it on the device by
     default, and only writes to the server when the build asked for
     cross-device. That is why there is no fetch here — a generated app makes no
     network requests at all, which is a smaller attack surface than trusting it
     to call only its own endpoint. */
  function flush() {
    if (!dirty) return;
    dirty = false;
    clearTimeout(timer);
    if (window.parent && window.parent !== window) {
      try { window.parent.postMessage({ type: 'henway:state', data: mem }, '*'); } catch (e) {}
    }
  }

  var shim = {
    getItem: function (k) {
      k = String(k);
      return Object.prototype.hasOwnProperty.call(mem, k) ? mem[k] : null;
    },
    setItem: function (k, v) { mem[String(k)] = String(v); schedule(); },
    removeItem: function (k) { delete mem[String(k)]; schedule(); },
    clear: function () { mem = {}; schedule(); },
    key: function (i) { var ks = Object.keys(mem); return i < ks.length ? ks[i] : null; }
  };
  Object.defineProperty(shim, 'length', { get: function () { return Object.keys(mem).length; } });

  /* defineProperty shadows the throwing prototype accessor with an own property
     on window. Verified working in the sandbox. */
  try {
    Object.defineProperty(window, 'localStorage',
      { value: shim, configurable: true, writable: true });
    Object.defineProperty(window, 'sessionStorage',
      { value: shim, configurable: true, writable: true });
  } catch (e) {}

  /* Last chance to hand it over. A debounce alone loses the final edit. */
  var hosted = !!(window.parent && window.parent !== window);
  window.addEventListener('pagehide', flush);
  document.addEventListener('visibilitychange', function () {
    if (document.visibilityState === 'hidden') flush();
  });

  return {
    mode: hosted ? 'hosted' : 'memory',
    backed: hosted,
    flush: flush,
    all: function () { var c = {}; for (var q in mem) c[q] = mem[q]; return c; }
  };
})();

  function splitLine(line) {
    if (line.indexOf('\t') > -1) return line.split('\t').map(function (c) { return c.trim(); });
    var out = [], cur = '', q = false;
    for (var i = 0; i < line.length; i++) {
      var ch = line[i];
      if (ch === '"') { if (q && line[i + 1] === '"') { cur += '"'; i++; } else q = !q; }
      else if (ch === ',' && !q) { out.push(cur.trim()); cur = ''; }
      else cur += ch;
    }
    out.push(cur.trim());
    return out;
  }

  function keyify(s) {
    return String(s).toLowerCase().trim().replace(/[^a-z0-9]+/g, '_').replace(/^_|_$/g, '');
  }

  function parsePaste(text) {
    var lines = text.split(/\r?\n/).filter(function (l) { return l.trim() !== ''; });
    if (!lines.length) return { rows: [], skipped: 0 };
    var first = splitLine(lines[0]);
    var header = null;
    var hasEmailWord = first.some(function (c) { return /e-?mail/i.test(c); });
    var firstHasAddress = first.some(function (c) { return isEmail(c); });
    if (hasEmailWord && !firstHasAddress) { header = first.map(keyify); lines.shift(); }

    var rows = [], skipped = 0;
    lines.forEach(function (line) {
      var cells = splitLine(line);
      var email = '', name = '', extra = {};
      var emailIdx = -1;
      for (var i = 0; i < cells.length; i++) { if (isEmail(cells[i])) { email = cells[i]; emailIdx = i; break; } }
      if (!email) { skipped++; return; }
      if (header) {
        header.forEach(function (h, i) {
          var v = cells[i] || '';
          if (/^e_?mail$/.test(h) || h === 'email_address') return;
          if (h === 'name' || h === 'full_name' || h === 'first_name') { if (!name) name = v; return; }
          if (h && v) extra[h] = v;
        });
      }
      if (!name) {
        for (var j = 0; j < cells.length; j++) { if (j !== emailIdx && cells[j]) { name = cells[j]; break; } }
      }
      if (!name) name = email.split('@')[0];
      rows.push({
        id: 'r' + Date.now().toString(36) + Math.random().toString(36).slice(2, 7),
        name: name, email: email, extra: extra, status: 'pending', marked: null, example: false
      });
    });
    return { rows: rows, skipped: skipped };
  }

window.henwayEmail = function (o) {
  o = o || {};

  if (o.body) q.push('body=' + encodeURIComponent(o.body));
  if (o.cc) q.push('cc=' + encodeURIComponent(o.cc));

     silently dropped — the button looked completely dead, with no error. The
     shell is ours and is not sandboxed, so it opens the mail app on our
     behalf. Widening the sandbox instead would hand every generated app the

     a mailto link. */
  if (window.parent && window.parent !== window) {
    try { window.parent.postMessage({ type: 'henway:mailto', href: href }, '*'); return true; }

/* henwayBackup(key) / henwayRestore(key, cb) — the answer to the one limit that
   actually burns people: a self-contained app keeps data in ONE browser, and
   clearing site data loses it. This is not sync; it is a file the user owns and
   can move to another device or keep as insurance. Honest, and needs no server. */
window.henwayBackup = function (key, filename) {
  var payload = {};
  try {
    if (key) { payload[key] = localStorage.getItem(key); }
    else { for (var i = 0; i < localStorage.length; i++) {
      var k = localStorage.key(i); payload[k] = localStorage.getItem(k); } }
  } catch (e) { return false; }
  var doc = { henway: 1, savedAt: new Date().toISOString(), data: payload };
  /* Same iOS fact as the calendar and CSV: an in-frame blob click never
     reaches the user. Framed, the holder saves it from a real origin. The
     shell downgrades non-text/* mimes to text/plain; the .json extension and
     the content are what henwayRestore reads, so nothing is lost. */
  if (window.parent && window.parent !== window) {
    try {
      window.parent.postMessage({ type: 'henway:download',
        text: JSON.stringify(doc, null, 2),
        filename: (filename || 'my-data') + '.json',
        mime: 'text/plain;charset=utf-8' }, '*');
      return true;
    } catch (e) { /* fall through */ }
  }
  var blob = new Blob([JSON.stringify(doc, null, 2)], { type: 'application/json' });
  var a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = (filename || 'my-data') + '.json';
  document.body.appendChild(a); a.click();
  /* 120s, not 0. iOS Safari fetches the blob AFTER click returns, so revoking
     on the next tick cancels the download before it exists — measured on the
     probe, 2026-08-31: returned true, no file, no error. The gesture was never
     the problem; this line was killing its own file. */
  setTimeout(function () { URL.revokeObjectURL(a.href); a.remove(); }, 120000);
  return true;
};
window.henwayRestore = function (cb) {
  var inp = document.createElement('input');
  inp.type = 'file'; inp.accept = 'application/json,.json';
  inp.addEventListener('change', function () {
    var f = inp.files && inp.files[0];
    if (!f) { cb && cb(false, 'No file chosen'); return; }
    var r = new FileReader();
    r.onload = function () {
      try {
        var doc = JSON.parse(r.result);
        if (!doc || !doc.data) { cb && cb(false, 'That is not a backup file'); return; }
        Object.keys(doc.data).forEach(function (k) {
          if (doc.data[k] !== null) localStorage.setItem(k, doc.data[k]);
        });
        cb && cb(true, doc.savedAt || '');
      } catch (e) { cb && cb(false, 'That file could not be read'); }
    };
    r.readAsText(f);
  });
  inp.click();
  return true;
};

/* henwayCSV(rows, filename) — rows is an array of arrays, or of objects with
   matching keys. Downloads a spreadsheet-ready file. */
window.henwayCSV = function (rows, filename) {
  if (!rows || !rows.length) return false;
  var data = rows;
  if (!Array.isArray(rows[0])) {
    var keys = Object.keys(rows[0]);
    data = [keys].concat(rows.map(function (r) {
      return keys.map(function (k) { return r[k]; });
    }));
  }
  var csv = data.map(function (r) {
    return r.map(function (c) {
      c = (c == null ? '' : String(c));
      return /[",\n]/.test(c) ? '"' + c.replace(/"/g, '""') + '"' : c;
    }).join(',');
  }).join('\r\n');
  /* The lesson the calendar snippet above already carries, verbatim: a blob
     download from inside a sandboxed, opaque-origin frame does not reach the
     user on iOS Safari — the click happens, nothing downloads, no error. This
     snippet was written without it; Michael's probe caught it returning true
     with nothing in hand, twice, through two wrong fixes. Framed, the file
     goes to the holder, whose "Tap to save" link is a real gesture on a real
     origin. */
  if (window.parent && window.parent !== window) {
    try {
      window.parent.postMessage({ type: 'henway:download', text: '\ufeff' + csv,
        filename: (filename || 'export') + '.csv',
        mime: 'text/csv;charset=utf-8' }, '*');
      return true;
    } catch (e) { /* fall through */ }
  }
  var blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8' });
  var a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = (filename || 'export') + '.csv';
  document.body.appendChild(a); a.click();
  /* 120s, not 0. iOS Safari fetches the blob AFTER click returns, so revoking
     on the next tick cancels the download before it exists — measured on the
     probe, 2026-08-31: returned true, no file, no error. The gesture was never
     the problem; this line was killing its own file. */
  setTimeout(function () { URL.revokeObjectURL(a.href); a.remove(); }, 120000);
  return true;
};
